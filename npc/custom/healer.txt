//===== rAthena Script =======================================
//= Healer (Pre-Renewal Optimized)
//===== Description: =========================================
//= Fast, configurable Pre-Renewal healer with optional buffs,
//= debuff removal, broken gear repair, and auto-identify.
//============================================================

-	script	Healer	-1,{

	// Cooldown check
	if (.Delay && @HD > gettimetick(2)) {
		dispbottom "Healer: Please wait " + (@HD - gettimetick(2)) + " second(s) before healing again.";
		end;
	}

	// Calculate fee (free for low-level players if .FreeLevel is set)
	.@cost = (.Price > 0 && BaseLevel > .FreeLevel) ? .Price : 0;

	if (.@cost > 0) {
		message strcharinfo(0), "Healing costs " + callfunc("F_InsertComma", .@cost) + " Zeny.";
		if (Zeny < .@cost) {
			dispbottom "Healer: You do not have enough Zeny.";
			end;
		}
		if (select("^0055FFHeal (" + callfunc("F_InsertComma", .@cost) + "z)^000000:^777777Cancel^000000") == 2)
			end;
		Zeny -= .@cost;
	}

	// Restore 100% HP & SP
	specialeffect2 EF_HEAL2;
	percentheal 100, 100;

	// Cure negative Pre-Renewal status ailments
	if (.ClearDebuffs) {
		sc_end SC_POISON;
		sc_end SC_SILENCE;
		sc_end SC_BLIND;
		sc_end SC_CONFUSION;
		sc_end SC_CURSE;
		sc_end SC_HALLUCINATION;
		sc_end SC_BLEEDING;
		sc_end SC_DECREASEAGI;
	}

	// Repair all broken equipment automatically
	if (.RepairAll && getbrokenid(1)) {
		repairall;
		specialeffect2 EF_REPAIRWEAPON;
		dispbottom "Healer: Your broken equipment has been repaired.";
	}

	// Identify all unidentified items in inventory automatically
	if (.IdentifyAll) {
		.@id_count = identifyall();
		if (.@id_count > 0)
			dispbottom "Healer: Identified " + .@id_count + " item(s).";
	}

	// Apply Pre-Renewal support buffs
	if (.Buffs) {
		specialeffect2 EF_INCAGILITY;
		sc_start SC_INCREASEAGI, .BuffDuration, 10;
		specialeffect2 EF_BLESSING;
		sc_start SC_BLESSING, .BuffDuration, 10;

		if (.ExtraBuffs) {
			sc_start SC_MAGNIFICAT, .BuffDuration, 5;
			sc_start SC_GLORIA, .BuffDuration, 5;
		}
	}

	// Set cooldown timestamp
	if (.Delay)
		@HD = gettimetick(2) + .Delay;

	end;

OnInit:
	.Price = 0;               // Zeny cost per heal (0 = free)
	.FreeLevel = 70;          // Free healing up to this Base Level when .Price > 0
	.Buffs = 1;               // Grant Blessing Lv.10 & Increase AGI Lv.10? (1 = yes, 0 = no)
	.ExtraBuffs = 0;          // Also grant Magnificat Lv.5 & Gloria Lv.5? (1 = yes, 0 = no)
	.BuffDuration = 240000;   // Buff duration in milliseconds (240000 = 4 minutes)
	.ClearDebuffs = 1;        // Cure Poison, Silence, Blind, Curse, etc.? (1 = yes, 0 = no)
	.RepairAll = 1;           // Automatically repair broken equipment? (1 = yes, 0 = no)
	.IdentifyAll = 1;         // Automatically identify all unidentified items? (1 = yes, 0 = no)
	.Delay = 0;               // Cooldown between heals in seconds (0 = no delay)
	end;
}
