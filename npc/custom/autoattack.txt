//===== rAthena Script =======================================
//= Auto Attack (menu version)
//===== Version: =============================================
//= 2.0
//===== Description: =========================================
//= Players click the "Auto Attack Menu" item (drag it to the
//= hotbar) or type @aa to open a settings menu, then start a
//= loop that targets monsters around them.
//=   @aa      = open the menu
//=   @aa off  = stop immediately
//===== Install: =============================================
//= - this file : npc/custom/autoattack.txt
//=   add  npc: npc/custom/autoattack.txt  to npc/scripts_custom.conf
//= - the item  : see autoattack_item.yml (db/import/item_db.yml)
//============================================================

-	script	AutoAttack	-1,{
	end;

OnInit:
	// ------------------------- CONFIG -------------------------
	.item_id = 32100;            // must match "Id" in item_db
	.give_item = 1;              // 1 = give the menu item at login if the player has none
	.max_range = 14;             // largest range (cells) a player may set
	.default_range = 7;          // range for first-time users
	.tick = 400;                 // ms between checks
	.stuck_ticks = 10;           // ticks without hurting the target before it is skipped
	.max_weight_pct = 90;        // stop at this % weight (0 = off)
	.block_towns = 1;            // 1 = Auto Attack cannot run in town maps
	.debug = 1;                 // 1 = print "[AA debug]" lines in the chat window (set 0 once it works)
	setarray .ignore_ids, 1288;  // monster IDs that are never targeted (1288 = Emperium)
	// ----------------------------------------------------------
	bindatcmd "aa", "AutoAttack::OnAtcommand";
	bindatcmd "autoattack", "AutoAttack::OnAtcommand";
	end;

OnPCLoginEvent:
	if (.give_item && !countitem(.item_id) && getitemname(.item_id) != "null")
		getitem .item_id, 1;
	end;

// Item click (doevent from the item script) and @aa both land here
OnItemClick:
OnAtcommand:
	if (.@atcmd_parameters$[0] == "off") {
		@aa_on = 0;
		dispbottom "[Auto Attack] Stopped.";
		end;
	}
	callsub S_Menu;
	end;

// ------------------------------------------------------------
// Settings menu
// ------------------------------------------------------------
S_Menu:
	if (aa_range < 1 || aa_range > .max_range) aa_range = .default_range;
	if (aa_mode < 1 || aa_mode > 2) aa_mode = 1;
	while (1) {
		if (.@drawn) clear;   // 'clear' needs an open dialog, so skip it on the first pass
		.@drawn = 1;
		mes "[ ^0055FFAuto Attack^000000 ]";
		mes "Status: " + (@aa_on ? "^00AA00RUNNING^000000" : "^FF0000STOPPED^000000");
		mes "Target: " + (aa_mode == 1 ? "Nearest monster" : "Lowest HP monster");
		mes "Range: " + aa_range + " cells";
		mes "Stop at HP: " + (aa_hpstop ? aa_hpstop + "%" : "Off");
		mes "Auto-loot: " + (aa_loot ? "On" : "Off");
		.@s = select((@aa_on ? "Stop Auto Attack" : "Start Auto Attack") + ":Target mode:Attack range:Stop at low HP:Toggle auto-loot:Close");

		if (.@s == 1) {
			if (@aa_on) {
				@aa_on = 0;
				dispbottom "[Auto Attack] Stopped.";
			} else if (.block_towns && getmapflag(strcharinfo(3), mf_town)) {   // strcharinfo type 3 = map name
				dispbottom "[Auto Attack] Can't be used in towns.";
			} else {
				@aa_on = 1;
				@aa_run++;
				if (aa_loot) atcommand "@autoloot 100";
				close2;
				callsub S_Loop, @aa_run;
				end;
			}
		} else if (.@s == 2) {
			clear;
			mes "[ ^0055FFTarget mode^000000 ]";
			mes "Nearest: attack the closest monster.";
			mes "Lowest HP: finish weak monsters first.";
			.@m = select("Nearest monster:Lowest HP monster:Back");
			if (.@m < 3) aa_mode = .@m;
		} else if (.@s == 3) {
			clear;
			mes "[ ^0055FFAttack range^000000 ]";
			mes "Monsters farther away than this are ignored.";
			mes "Current: " + aa_range + " cells (max " + .max_range + ")";
			.@r = select("3 cells (melee):5 cells:7 cells:10 cells:Maximum:Type a number:Back");
			switch (.@r) {
			case 1: aa_range = 3; break;
			case 2: aa_range = 5; break;
			case 3: aa_range = 7; break;
			case 4: aa_range = 10; break;
			case 5: aa_range = .max_range; break;
			case 6:
				mes "Enter a number from 1 to " + .max_range + ".";
				input .@v, 1, .max_range;
				aa_range = .@v;
				break;
			}
			if (aa_range > .max_range) aa_range = .max_range;
		} else if (.@s == 4) {
			clear;
			mes "[ ^0055FFStop at low HP^000000 ]";
			mes "Auto Attack stops when your HP drops to this percent.";
			.@h = select("Off:20%:30%:50%:70%:Back");
			switch (.@h) {
			case 1: aa_hpstop = 0; break;
			case 2: aa_hpstop = 20; break;
			case 3: aa_hpstop = 30; break;
			case 4: aa_hpstop = 50; break;
			case 5: aa_hpstop = 70; break;
			}
		} else if (.@s == 5) {
			aa_loot = !aa_loot;
			if (aa_loot) atcommand "@autoloot 100";
			else atcommand "@autoloot 0";
		} else {
			close;
		}
	}
	end;

// ------------------------------------------------------------
// Attack loop. Runs attached to the player; sleep2 keeps the player attached.
// ------------------------------------------------------------
S_Loop:
	.@run = getarg(0);
	.@cur = 0;
	.@cur_hp = 0;
	.@same = 0;
	.@t = 0;
	dispbottom "[Auto Attack] Started. Use @aa off to stop.";
	if (.debug) dispbottom "[AA debug] loop started: range=" + aa_range + " mode=" + aa_mode;
	while (1) {
		sleep2 .tick;
		if (@aa_on != 1 || @aa_run != .@run) end;   // stopped, or replaced by a newer loop
		.@dbg++;

		// ---- safety checks ----
		if (Hp <= 0) {
			callsub S_Stop, "you died.";
			end;
		}
		.@cv = checkvending();
		.@cc = checkchatting();
		if (.@cv || .@cc) {
			if (.debug && .@dbg % 5 == 0) dispbottom "[AA debug] paused: vending=" + .@cv + " chatting=" + .@cc;
			continue;
		}
		getmapxy(.@map$, .@px, .@py);   // no type given = the attached player
		if (.block_towns && getmapflag(.@map$, mf_town)) {
			callsub S_Stop, "not allowed in towns.";
			end;
		}
		if (aa_hpstop && Hp * 100 / MaxHp <= aa_hpstop) {
			callsub S_Stop, "HP is low.";
			end;
		}
		if (.max_weight_pct && Weight * 100 / MaxWeight >= .max_weight_pct) {
			callsub S_Stop, "you are too heavy.";
			end;
		}

		.@t++;
		if (.@t >= 100) {            // forget skipped monsters every ~40 seconds
			.@t = 0;
			deletearray .@skip;
		}
		.@r = aa_range;

		// ---- keep the current target while it is alive, in range and getting hurt ----
		if (.@cur) {
			.@keep = 0;
			deletearray .@d;
			getunitdata .@cur, .@d;
			if (getarraysize(.@d)) {
				if (.@d[UMOB_HP] > 0 && distance(.@px, .@py, .@d[UMOB_X], .@d[UMOB_Y]) <= .@r) {
					.@keep = 1;
					if (.@d[UMOB_HP] >= .@cur_hp) .@same++;
					else .@same = 0;
					.@cur_hp = .@d[UMOB_HP];
					if (.@same >= .stuck_ticks) {        // blocked / unreachable / immune
						.@skip[getarraysize(.@skip)] = .@cur;
						if (.debug) dispbottom "[AA debug] target " + .@cur + " is not getting hurt, skipping it";
						.@keep = 0;
					}
				}
			}
			if (.@keep) continue;
			.@cur = 0;
			.@same = 0;
		}

		// ---- pick a new target ----
		.@best = 0;
		.@n_skip = 0;
		.@n_nodata = 0;
		.@n_dead = 0;
		.@n_ign = 0;
		.@n_far = 0;
		.@n_ok = 0;
		.@n = getareaunits(BL_MOB, .@map$, .@px - .@r, .@py - .@r, .@px + .@r, .@py + .@r, .@gids[0]);
		for (.@i = 0; .@i < .@n; .@i++) {
			.@g = .@gids[.@i];
			if (inarray(.@skip, .@g) >= 0) {
				.@n_skip++;
				continue;
			}
			deletearray .@d;
			getunitdata .@g, .@d;
			if (!getarraysize(.@d)) {
				.@n_nodata++;
				continue;
			}
			if (.@d[UMOB_HP] <= 0) {
				.@n_dead++;
				continue;
			}
			if (inarray(.ignore_ids, .@d[UMOB_CLASS]) >= 0) {
				.@n_ign++;
				continue;
			}
			.@dist = distance(.@px, .@py, .@d[UMOB_X], .@d[UMOB_Y]);
			if (.@dist > .@r) {
				.@n_far++;
				continue;
			}
			.@n_ok++;
			.@score = (aa_mode == 2 ? .@d[UMOB_HP] * 100 + .@dist : .@dist);
			if (!.@best || .@score < .@best_score) {
				.@best = .@g;
				.@best_score = .@score;
				.@best_hp = .@d[UMOB_HP];
			}
		}
		if (.debug && (.@best || .@dbg % 5 == 0)) {
			dispbottom "[AA debug] " + .@map$ + " " + .@px + "," + .@py + " r=" + .@r + " scan=" + .@n + " skip=" + .@n_skip + " nodata=" + .@n_nodata + " dead=" + .@n_dead + " ign=" + .@n_ign + " far=" + .@n_far + " ok=" + .@n_ok;
			if (.@n) dispbottom "[AA debug] last unit: gid=" + .@g + " size=" + getarraysize(.@d) + " hp=" + .@d[UMOB_HP] + " x=" + .@d[UMOB_X] + " y=" + .@d[UMOB_Y] + " class=" + .@d[UMOB_CLASS];
		}
		if (!.@best) continue;

		// 1 = continuous attack (players only repeat attacks when this is non-zero)
		if (.debug) dispbottom "[AA debug] attacking gid=" + .@best + " hp=" + .@best_hp;
		unitattack getcharid(3), .@best, 1;
		.@cur = .@best;
		.@cur_hp = .@best_hp;
		.@same = 0;
	}
	end;

S_Stop:
	@aa_on = 0;
	dispbottom "[Auto Attack] Stopped: " + getarg(0);
	return;
}
